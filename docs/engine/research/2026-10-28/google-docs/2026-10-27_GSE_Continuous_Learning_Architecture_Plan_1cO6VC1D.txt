Contract: artifacts/telemetry-schema-2026-10-27.md & artifacts/rag-telemetry-wiring-2026-10-27.md

"""
import argparse
import datetime
import hashlib
import json
import os
import uuid
from pathlib import Path
from typing import Any, Dict, List, Optional

EXPECTED_SEALED_HASH = "85b5cd3303c540492b23019d913aabd9179945a72c490579124e6e57dcad68b8"
FROZEN_CONDITIONS_ON = {
   "version": "v1.0",
   "tau": 0.42,
   "qb_questionable_epa": 0.0947,

VALID_SOURCE_TYPES = {"injury_note", "practice_report", "dk_line", "beat_writer", "form4", "arxiv", "other"}
VALID_STATUSES = {"ingested", "retrieved", "used_in_overlay", "rejected"}


def validate_delta_record(rec: Dict[str, Any], active_sealed_hash: str = EXPECTED_SEALED_HASH) -> None:
   for req_key in ("id", "timestamp", "source", "claim", "confidence", "sealed_hash", "conditions_on", "status", "notes"):
       if req_key not in rec:
           raise ValueError(f"SCHEMA_VIOLATION: Missing required key '{req_key}'")
   src = rec["source"]
   if src.get("type") not in VALID_SOURCE_TYPES:
       raise ValueError(f"SCHEMA_VIOLATION: Invalid source.type '{src.get('type')}'")
   if not isinstance(src.get("url"), str) or not src["url"].startswith("http"):
       raise ValueError("SCHEMA_VIOLATION: source.url must be a valid URL string")
   if not isinstance(src.get("raw_excerpt"), str) or not src["raw_excerpt"].strip():
       raise ValueError("SCHEMA_VIOLATION: source.raw_excerpt must be non-empty verbatim text")
   clm = rec["claim"]
   for c_key in ("subject", "predicate", "object", "game_id", "week"):
       if c_key not in clm:
           raise ValueError(f"SCHEMA_VIOLATION: Missing claim.{c_key}")
   conf = rec["confidence"]
   if not isinstance(conf, (int, float)) or not (0.0 <= float(conf) <= 1.0):
       raise ValueError(f"SCHEMA_VIOLATION: confidence {conf} outside [0.0, 1.0]")
   if rec["sealed_hash"] != active_sealed_hash:
       raise ValueError(f"SEAL_HASH_MISMATCH: record sealed_hash {rec['sealed_hash']} != {active_sealed_hash}")
   cond = rec["conditions_on"]
   if (
       cond.get("version") != "v1.0"
       or float(cond.get("tau", -1.0)) != 0.42
       or float(cond.get("qb_questionable_epa", -1.0)) != 0.0947
   ):
       raise ValueError(f"CONDITIONS_ON_VIOLATION: {cond} does not match frozen Tuesday v1.0 parameters")
   if rec["status"] not in VALID_STATUSES:
       raise ValueError(f"SCHEMA_VIOLATION: Invalid status '{rec['status']}'")


def append_delta_record(delta_file_path: str, record: Dict[str, Any], active_sealed_hash: str = EXPECTED_SEALED_HASH) -> str:
   validate_delta_record(record, active_sealed_hash)
   path_obj = Path(delta_file_path)
   path_obj.parent.mkdir(parents=True, exist_ok=True)
   if path_obj.exists():
       with open(path_obj, "r", encoding="utf-8") as f:
           for line in f:
               line = line.strip()
               if not line:
                   continue
               existing = json.loads(line)
               if existing.get("id") == record["id"]:
                   raise RuntimeError(f"NO_OVERWRITE_VIOLATION: Record id {record['id']} already exists in {delta_file_path}")
   with open(path_obj, "a", encoding="utf-8") as f:
       f.write(json.dumps(record, separators=(",", ":")) + "\n")
   return str(path_obj)


def retrieve_prior_claims(
   delta_file_path: str,
   sealed_hash: str,
   subject: Optional[str] = None,
   game_id: Optional[str] = None,
   allowed_source_types: Optional[set] = None,
) -> List[Dict[str, Any]]:
   """
   Read-only retrieval of prior claims matching `sealed_hash` + `subject` (and optional `game_id`).
   Used as context before Sinkhorn overlay or Step N residual evaluation; never mutates sealed state.
   """
   path_obj = Path(delta_file_path)
   if not path_obj.exists():
       return []
   matches: List[Dict[str, Any]] = []
   with open(path_obj, "r", encoding="utf-8") as f:
       for line in f:
           line = line.strip()
           if not line:
               continue
           rec = json.loads(line)
           if rec.get("sealed_hash") != sealed_hash:
               continue
           if rec.get("status") == "rejected":
               continue
           if subject and rec.get("claim", {}).get("subject") != subject:
               continue
           if game_id and rec.get("claim", {}).get("game_id") != game_id:
               continue
           if allowed_source_types and rec.get("source", {}).get("type") not in allowed_source_types:
               continue
           matches.append(rec)
   matches.sort(key=lambda r: r["timestamp"])
   return matches

Area 1: Real-Time Multi-Source Ingestion Patterns That Preserve Provenance
Production intelligence systems that fuse high-velocity news, beat-writer social feeds, official team announcements, audio transcripts, market boards, and corporate filings prevent narrative contamination by separating append-only bi-temporal episodic logging from temporal knowledge graph state resolution1. Every ingested claim is tagged along two orthogonal time dimensions: valid time (  , the point or interval in physical time when an injury, practice participation state, or transaction actually occurred) and transaction time (  , the exact UTC timestamp when the source published and the engine recorded the assertion)1.
Within GSE, the foundational ingestion contract is already defined in packages/prediction-engine/src/feeds/emit-mind-feed-record.ts (GseMindFeedV2Record, schema version "gse-mind-feed-v2.0") and now bridged to artifacts/delta-YYYY-MM-DD.jsonl (artifacts/telemetry-schema-2026-10-27.md)1. Every document, email, transcript, or market event is written as an append-only JSONL entry into brain/feeds/<YYYY-MM-DD>/ containing record_id, source_type ("gdrive" | "gdocs" | "gmail" | "rag_anything" | "settlement"), doc_or_thread_id, revision_or_msg_id, modality ("text" | "image" | "table" | "equation"), valid_time_iso, transaction_time_iso, entity_id, graph_node_id, content_sha256, conflict_on_record: boolean, and payload: { text_or_latex, verbatim_quote_span: { start, end } }, while actionable player, team, game, or coordinator claims are simultaneously appended to artifacts/delta-YYYY-MM-DD.jsonl carrying sealed_hash and conditions_on: { version: "v1.0", tau: 0.42, qb_questionable_epa: 0.0947 }1. Both writers enforce a strict NO_OVERWRITE_VIOLATION check against existing IDs, block any direct write to mind.jsonl, and verify SHA-256 digests before flushing to disk1. For Google Drive and Google Docs, the poller queries the Drive v3 revisions.list API (drive.readonly) filtered by modifiedTime <= asOfT; for Gmail, the poller queries q = "subject:(GSE) OR label:ICEBOX_LONG_READ_LABEL after:2026/09/08 before:{asOfEpochSeconds}", grouping by threadId in ascending internalDate order and stripping quoted reply blocks via SHA-256 paragraph hash deduplication1.
To resolve evolving and contradictory claims over an NFL week without losing historical lineage, GSE's gse-mind-feed-v2.0 and delta-YYYY-MM-DD.jsonl streams pair directly with Zep / Graphiti (Rasmussen, Paliychuk, Beauvais, Ryan, & Chalef, 2025, "Zep: A Temporal Knowledge Graph Architecture for Agent Memory", arXiv:2501.13956, open-source framework at https://github.com/getzep/graphiti)4. Graphiti models memory as a dynamic temporal knowledge graph    comprising three hierarchical tiers: an Episode Subgraph    storing non-lossy raw data episodes (raw injury reports, beat-writer tweets, podcast segments), a Semantic Entity Subgraph    where extracted entities and facts link back to their source episodes via episodic provenance edges   , and a Community Subgraph    built via Leiden community detection16. Crucially, every relational fact    carries explicit bi-temporal validity windows    alongside transaction timestamps   4. When a beat writer tweets a speculative practice note on Wednesday and an official team transaction activates a player off IR on Saturday, Graphiti and retrieve_prior_claims() do not overwrite or delete the Wednesday record; they supersede the Wednesday state with the higher-confidence Saturday official transaction traced to its episode4. Furthermore, as demonstrated in Eywa: Provenance-Grounded Long-Term Memory for AI Agents (arXiv:2605.30771), keeping conflicting claims live with explicit provenance (conflict_on_record: true) rather than allowing an LLM to auto-resolve them at ingestion time prevents narrative smoothing and ensures deterministic downstream state queries (AS OF timestamp   ) always resolve a single canonical player state1.
