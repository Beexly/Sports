def resolve_runtime_sealed_hash(repo_root: Optional[str] = None) -> str:
   """
   Standing Rule 3: Resolve sealed_hash at runtime from disk. Never hardcode a digest literal.
   Checks Windows canonical candidates/sealed-center.json first, then Linux /opt/model/sealed_baseline.json.
   """
   root = Path(repo_root) if repo_root else Path(__file__).resolve().parents[2]
   win_seal_center = root / "candidates" / "sealed-center.json"
   if win_seal_center.exists():
       data = json.loads(win_seal_center.read_text(encoding="utf-8"))
       for key in ("merkle", "sealed_hash", "coachMindsetHash", "seal_hash"):
           if isinstance(data.get(key), str) and len(data[key]) >= 16:
               return data[key]
       blob = json.dumps(data, sort_keys=True, separators=(",", ":")).encode("utf-8")
       return hashlib.sha256(blob).hexdigest()

   linux_seal = Path("/opt/model/sealed_baseline.json")
   if linux_seal.exists():
       data = json.loads(linux_seal.read_text(encoding="utf-8"))
       blob = json.dumps(data, sort_keys=True, separators=(",", ":")).encode("utf-8")
       return hashlib.sha256(blob).hexdigest()

def validate_delta_record(rec: Dict[str, Any], active_sealed_hash: str) -> None:
   for req_key in ("id", "timestamp", "source", "claim", "confidence", "sealed_hash", "conditions_on", "status", "notes"):
       if req_key not in rec:
           raise ValueError(f"SCHEMA_VIOLATION: Missing required key '{req_key}'")
   src = rec["source"]
   if src.get("type") not in VALID_SOURCE_TYPES:
       raise ValueError(f"SCHEMA_VIOLATION: Invalid source.type '{src.get('type')}'")
   if not isinstance(src.get("url"), str) or not src["url"].startswith("http"):
       raise ValueError("SCHEMA_VIOLATION: source.url must be a valid URL string")
   if not isinstance(src.get("raw_excerpt"), str) or not src["raw_excerpt"].strip():
       raise ValueError("SCHEMA_VIOLATION: source.raw_excerpt must be non-empty verbatim text")
   clm = rec["claim"]
   for c_key in ("subject", "predicate", "object", "game_id", "week"):
       if c_key not in clm:
           raise ValueError(f"SCHEMA_VIOLATION: Missing claim.{c_key}")
   conf = rec["confidence"]
   if not isinstance(conf, (int, float)) or not (0.0 <= float(conf) <= 1.0):
       raise ValueError(f"SCHEMA_VIOLATION: confidence {conf} outside [0.0, 1.0]")
   if rec["sealed_hash"] != active_sealed_hash:
       raise ValueError(f"SEAL_HASH_MISMATCH: record sealed_hash {rec['sealed_hash']} != {active_sealed_hash}")
   cond = rec["conditions_on"]
   if (
       cond.get("version") != "v1.0"
       or float(cond.get("tau", -1.0)) != 0.42
       or float(cond.get("qb_questionable_epa", -1.0)) != 0.0947
   ):
       raise ValueError(f"CONDITIONS_ON_VIOLATION: {cond} does not match frozen Tuesday v1.0 parameters")
   if rec["status"] not in VALID_STATUSES:
       raise ValueError(f"SCHEMA_VIOLATION: Invalid status '{rec['status']}'")
   if src.get("type") in CLASS_B_ONLY_SOURCES and rec["status"] == "used_in_overlay":
       raise ValueError(f"CLASS_A_FIREWALL_VIOLATION: Class B source '{src.get('type')}' cannot be used_in_overlay")


def append_delta_record(delta_file_path: str, record: Dict[str, Any], active_sealed_hash: str) -> str:
   validate_delta_record(record, active_sealed_hash)
   path_obj = Path(delta_file_path)
   path_obj.parent.mkdir(parents=True, exist_ok=True)
   if path_obj.exists():
       with open(path_obj, "r", encoding="utf-8") as f:
           for line in f:
               line = line.strip()
               if not line:
                   continue
               existing = json.loads(line)
               if existing.get("id") == record["id"]:
                   raise RuntimeError(f"NO_OVERWRITE_VIOLATION: Record id {record['id']} already exists in {delta_file_path}")
   with open(path_obj, "a", encoding="utf-8") as f:
       f.write(json.dumps(record, separators=(",", ":")) + "\n")
       f.flush()
       os.fsync(f.fileno())
   return str(path_obj)


def retrieve_prior_claims(
   delta_file_path: str,
   sealed_hash: str,
   subject: Optional[str] = None,
   game_id: Optional[str] = None,
   allowed_source_types: Optional[set] = None,
) -> List[Dict[str, Any]]:
   """
   Read-only retrieval of prior claims matching `sealed_hash` + `subject` (and optional `game_id`).
   Enforces bi-temporal entity reduction so latest official status supersedes earlier tags.
   """
   path_obj = Path(delta_file_path)
   if not path_obj.exists():
       return []
   matches: List[Dict[str, Any]] = []
   with open(path_obj, "r", encoding="utf-8") as f:
       for line in f:
           line = line.strip()
           if not line:
               continue
           rec = json.loads(line)
           if rec.get("sealed_hash") != sealed_hash:
               continue
           if rec.get("status") == "rejected":
               continue
           if subject and rec.get("claim", {}).get("subject") != subject:
               continue
           if game_id and rec.get("claim", {}).get("game_id") != game_id:
               continue
           if allowed_source_types and rec.get("source", {}).get("type") not in allowed_source_types:
               continue
           matches.append(rec)
   matches.sort(key=lambda r: r["timestamp"])
   return matches

       if (np.sum(np.abs(P.sum(axis=1) - r)) + np.sum(np.abs(P.sum(axis=0) - c))) <= 1e-9:
           break
   feasible_shares = P.sum(axis=0)
   mass_moved = float(0.5 * np.sum(np.abs(feasible_shares - r)))
   return feasible_shares, mass_moved


def simulate_matchup_pbp(away: str, home: str, away_inj: list, home_inj: list, sealed_state, expected_hash: str, n_sims: int = 5000, seed: int = 42) -> dict:

   home_eff_mult = sum(home_sh[i] * (exec_mults.get(rn, 0.90) if rn in home_inj else 1.0) for i, rn in enumerate(role_names))
   away_drop = sum(float(epa_table[fl]) for fl in away_inj if fl in epa_table)
   home_drop = sum(float(epa_table[fl]) for fl in home_inj if fl in epa_table)
   away_net_epa = ((tr_away["off_epa"] - away_drop) * away_eff_mult) - tr_home["def_epa"]
   home_net_epa = ((tr_home["off_epa"] - home_drop) * home_eff_mult) - tr_away["def_epa"] + (tr_home["hfa"] / 79.1)

           if np.any(is_to):

               g_nt = yds[non_to]
               new_yl = yardline_100[nt_i] - g_nt
               td_nt = new_yl <= 0
               if np.any(td_nt):
                   td_i = nt_i[td_nt]
                   home_score[td_i[poss[td_i] == 1]] += 6.95
