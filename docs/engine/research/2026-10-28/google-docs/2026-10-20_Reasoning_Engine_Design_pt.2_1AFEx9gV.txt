export interface ResidualContrastStats {

  readonly controlMean: number;

  readonly nControl: number;

export interface MediatorExpansionAttempt {
  readonly edge: readonly [NodeId, NodeId];

/**
 * VanderWeele & Ding (2017, Annals of Internal Medicine 167(4):268-274, Table 1 & Table 2):
 * Computes point and 95% CI lower-bound E-values from standardized difference d = (mu1 - mu0) / sd
 * via RR ≈ exp(0.91 * |d|), or directly from rate ratios. Returns null if sample size < 30 per arm.
 */
export function computeMeasuredEValueFromResiduals(
  treatedResiduals: readonly number[],
  controlResiduals: readonly number[]
): ResidualContrastStats | null {
  const n1 = treatedResiduals.length;
  const n0 = controlResiduals.length;
  if (n1 < 30 || n0 < 30) return null;


  const mean1 = treatedResiduals.reduce((a, b) => a + b, 0) / n1;
  const mean0 = controlResiduals.reduce((a, b) => a + b, 0) / n0;
  const var1 = treatedResiduals.reduce((acc, v) => acc + (v - mean1) ** 2, 0) / (n1 - 1);
  const var0 = controlResiduals.reduce((acc, v) => acc + (v - mean0) ** 2, 0) / (n0 - 1);
  const pooledSd = Math.sqrt(((n1 - 1) * var1 + (n0 - 1) * var0) / (n1 + n0 - 2));


  if (!Number.isFinite(pooledSd) || pooledSd <= 1e-8) return null;


  const d = Math.abs(mean1 - mean0) / pooledSd;
  const seD = Math.sqrt((n1 + n0) / (n1 * n0) + (d * d) / (2 * (n1 + n0)));
  const dLower = Math.max(0, d - 1.96 * seD);


  const rrPoint = Math.exp(0.91 * d);
  const rrLower = Math.exp(0.91 * dLower);


  const eValuePoint = rrPoint + Math.sqrt(rrPoint * (rrPoint - 1));
  const eValueCi = rrLower <= 1.0 ? 1.0 : rrLower + Math.sqrt(rrLower * (rrLower - 1));


  return {
    treatedMean: mean1,
    controlMean: mean0,
    pooledSd,
    nTreated: n1,
    nControl: n0,
    pointRiskRatio: rrPoint,
    ciLowerRiskRatio: rrLower,
    measuredEValue: eValuePoint,
    measuredCiEValue: eValueCi,
  };
}


/**
 * Expand directed edge (u -> v) with mediator M: replaces (u -> v) with (u -> M) and (M -> v)
 * with zero incident bidirected edges on M (Theorem 2 & Algorithm 2 of Väänänen et al. 2025, arXiv:2610.08092v2).
 */
export function expandDirectedEdgeWithMediator(
  g: CausalADMG,
  edge: readonly [NodeId, NodeId],
  mediator: NodeId
): CausalADMG {
  const [u, v] = edge;
  const uIdx = g.nodes.indexOf(u);
  const vIdx = g.nodes.indexOf(v);
  const newNodes = [...g.nodes];
  if (!newNodes.includes(mediator)) {
    const insertIdx = uIdx >= 0 && vIdx > uIdx ? vIdx : newNodes.length;
    newNodes.splice(insertIdx, 0, mediator);
  }
  const newDirected = g.directed.filter(([a, b]) => !(a === u && b === v));
  newDirected.push([u, mediator], [mediator, v]);
  return {
    nodes: newNodes,
    directed: newDirected,
    bidirected: [...g.bidirected],
  };
}

      status: "IDENTIFIED_APPROVED",
      refusalCode: "NONE",
      measuredEValue: eVal,
      activeGraph: input.admg,
      expansionAttempts: [],
    };
  }


  // Step 3: Base ID encountered a hedge -> Run HCID + planHedgeBreakage with synchronous E-value measurement
  const attempts = planHedgeBreakage(
    input.admg,
    input.ySet,
    input.xSet,
    baseId.enablingEdges,
    input.candidateMediators
  );


  const activatingAttempt = attempts.find((a) => a.activating === true);
  if (activatingAttempt) {
    const expandedGraph = expandDirectedEdgeWithMediator(
      input.admg,
      activatingAttempt.edge,
      activatingAttempt.mediatorNode
    );
    return {
      approved: true,
      status: "HEDGE_BROKEN_APPROVED",
      refusalCode: "NONE",
      measuredEValue: activatingAttempt.measuredEValue,
      activeGraph: expandedGraph,
      expansionAttempts: attempts,
    };
  }


  // Distinguish whether expansion broke the hedge graphically but failed the E-value gate vs. never broke the hedge
  const fragileBreak = attempts.find((a) => a.breaksHedgeButFragile === true);
  if (fragileBreak) {
    return {
      approved: false,
      status: "REFUSED_FRAGILE_E_VALUE",
      refusalCode: "REFUSED_G4_FRAGILE_E_VALUE_OR_PLACEBO",
      measuredEValue: fragileBreak.measuredEValue,
      activeGraph: input.admg,
      expansionAttempts: attempts,
    };
  }


  return {
    approved: false,
    status: "REFUSED_HEDGE",
    refusalCode: "REFUSED_G3_NON_IDENTIFIABLE_HEDGE",
    measuredEValue: null,
    activeGraph: input.admg,
    expansionAttempts: attempts,
  };
}
